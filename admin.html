async function loadStudents() { try { const response = await fetch('/api/students'); const students = await response.json(); let html = ''; students.forEach(s => { html += ` ${s.code} ${s.name} ${s.major} ${s.cohort} ${s.status === 1 ? 'Đang học' : 'Bị khóa'} `; }); document.getElementById('studentTableBody').innerHTML = html; } catch (error) { console.error("Lỗi tải danh sách sinh viên:", error); } }
